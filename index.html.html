<script>
// Data Level Tantangan (Bisa Anda ubah/tambah)
const levels = [
{
title: "Materi: Cara Membuat Mie Instan Sehat",
correctOrder: [
"Rebus air sebanyak 400 ml hingga mendidih.",
"Masukkan mie dan sayuran segar ke dalam air mendidih.",
"Tuangkan bumbu mie ke dalam mangkuk.",
"Tiriskan mie dan campurkan ke dalam mangkuk berisi bumbu.",
"Aduk rata dan sajikan mie selagi hangat."
]
},
{
title: "Materi: Cara Mematikan Komputer dengan Benar",
correctOrder: [
"Tutup semua aplikasi yang sedang digunakan.",
"Klik menu 'Start' pada pojok kiri bawah layar.",
"Pilih opsi 'Power' lalu klik 'Shut Down'.",
"Tunggu hingga layar monitor dan lampu tombol CPU mati total.",
"Cabut kabel listrik dari stopkontak jika sudah tidak digunakan."
]
}
];
let currentLevel = 0;
let lives = 3;
let score = 0;
let currentSteps = [];
function initLevel() {
if (currentLevel >= levels.length) {
document.querySelector('.game-card').innerHTML = `
<h2>🎉 Tantangan Selesai!</h2>
<p style="text-align:center;">Skor Akhir Anda: <strong>${score}</strong></p>
<p style="text-align:center;">Kemampuan awal Anda dalam memahami teks prosedur sangat
baik!</p>
`;
return;
}
document.getElementById('level-display').innerText = currentLevel + 1;
document.getElementById('instruction-text').innerText = levels[currentLevel].title;
document.getElementById('feedback-text').innerText = '';
// Acak urutan langkah untuk tantangan
currentSteps = [...levels[currentLevel].correctOrder].sort(() => Math.random() - 0.5);
renderSteps();
}
function renderSteps() {
const container = document.getElementById('step-container');
container.innerHTML = '';
currentSteps.forEach((step, index) => {
const li = document.createElement('li');
li.className = 'step-item';
li.innerHTML = `
<span>${step}</span>
<div class="btn-group">
<button onclick="moveStep(${index}, -1)">▲</button>
<button onclick="moveStep(${index}, 1)">▼</button>
</div>
`;
container.appendChild(li);
});
}
function moveStep(index, direction) {
const newIndex = index + direction;
if (newIndex >= 0 && newIndex < currentSteps.length) {
const temp = currentSteps[index];
currentSteps[index] = currentSteps[newIndex];
currentSteps[newIndex] = temp;
renderSteps();
}
}
function checkAnswer() {
const correct = levels[currentLevel].correctOrder;
let isCorrect = true;
for (let i = 0; i < correct.length; i++) {
if (currentSteps[i] !== correct[i]) {
isCorrect = false;
break;
}
}
const feedback = document.getElementById('feedback-text');
if (isCorrect) {
score += 50;
document.getElementById('score-display').innerText = score;
feedback.style.color = '#2ecc71';
feedback.innerText = 'Jawaban Benar! Melanjutkan ke tantangan berikutnya...';
currentLevel++;
setTimeout(initLevel, 1500);
} else {
lives--;
document.getElementById('lives-display').innerText = lives;
feedback.style.color = '#e74c3c';
if (lives > 0) {
feedback.innerText = 'Urutan masih belum tepat. Coba perbaiki lagi!';
} else {
document.querySelector('.game-card').innerHTML = `
<h2>❌ Game Over</h2>
<p style="text-align:center;">Nyawa Anda habis. Jangan berkecil hati, mari pelajari kembali
materi Teks Prosedur!</p>
`;
}
}
}
// Jalankan game saat pertama kali dimuat
initLevel();
</script>
